            Console.WriteLine("\n7. feladat: Statisztika");

            Dictionary<string, int> statisztika = new Dictionary<string, int>();

            foreach (var item in sportolok)
            {
                // Megnézzük, hogy az aktuális országkód benne van-e már a szótárban
                if (statisztika.ContainsKey(item.Országkód))
                {
                    // Ha igen, megnöveljük a darabszámot 1-gyel
                    statisztika[item.Országkód]++;
                }
                else
                {
                    // Ha nincs, akkor beletesszük 1-es kezdő darabszámmal
                    statisztika.Add(item.Országkód, 1);
                }
            }
